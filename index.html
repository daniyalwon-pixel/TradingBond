    저장 방식

    프로젝트 설정 정보 (이미 완료됨)

    1) supabase.com에서 프로젝트를 만듭니다. 2) SQL Editor에 아래 SQL을 붙여넣고 실행합니다. 3) Authentication → Users → Add user로 본인 이메일과 비밀번호를 만듭니다. 4) Project Settings → API의 Project URL과 anon public key를 아래에 입력하고 로그인합니다.

    create table journal (
  user_id uuid not null default auth.uid() references auth.users on delete cascade,
  date text not null,
  data jsonb not null,
  updated_at timestamptz default now(),
  primary key (user_id, date)
);
create table journal_img (
  id text primary key,
  user_id uuid not null default auth.uid() references auth.users on delete cascade,
  data text not null
);
alter table journal enable row level security;
alter table journal_img enable row level security;
create policy "own rows" on journal for all
  using (auth.uid() = user_id) with check (auth.uid() = user_id);
create policy "own rows" on journal_img for all
  using (auth.uid() = user_id) with check (auth.uid() = user_id);
